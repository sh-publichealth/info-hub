% tex/ian-house-style.tex
% ------------------------------------------------------------
% Ian House Style: Quarto → PDF (LaTeX) preamble
% Goals: clean typography, consistent header/footer, logo-ready,
%        and predictable behaviour in Quarto/Pandoc.
% ------------------------------------------------------------

% ---------- Core packages ----------
\usepackage{graphicx}      % images/logo
\usepackage{xcolor}        % colors
\usepackage{geometry}      % margins
\usepackage{microtype}     % subtle typographic improvements
\usepackage{setspace}      % line spacing
\usepackage{fancyhdr}      % header/footer control
\usepackage{lastpage}      % total page count
\usepackage{booktabs}      % nicer tables
\usepackage{longtable}     % long tables across pages
\usepackage{array}         % table column control
\usepackage{caption}       % caption spacing
\usepackage{hyperref}      % PDF links + metadata
\usepackage{enumitem}      % list spacing
\usepackage{titling}       % title-page hooks

% ---------- Page geometry ----------
% Default (all pages). You can override in YAML (geometry:) if you prefer.
\geometry{margin=1in}

% ---------- Typography ----------
\setstretch{1.08}

% ---------- Colors ----------
\definecolor{IanLinkBlue}{HTML}{045174}

% ---------- Hyperref setup ----------
\hypersetup{
  colorlinks=true,
  linkcolor=IanLinkBlue,
  urlcolor=IanLinkBlue,
  citecolor=IanLinkBlue,
  pdfborder={0 0 0}
}

% ---------- Captions ----------
\captionsetup{
  font=small,
  labelfont=bf,
  skip=6pt
}

% ---------- Table spacing ----------
\renewcommand{\arraystretch}{1.15}

% ---------- Lists ----------
\setlist{itemsep=2pt, topsep=4pt, parsep=0pt, partopsep=0pt}

% ---------- Safety: prevent overfull lines ----------
\emergencystretch=2em

% ============================================================
% Logos and labels
% ============================================================

% Small header logo (page 2+)
\newcommand{\IanLogoPath}{}       % e.g. {assets/brand/logo.png}
\newcommand{\IanLogoHeight}{14pt}

\newcommand{\IanMaybeLogo}{%
  \ifx\IanLogoPath\empty
    % no logo
  \else
    \includegraphics[height=\IanLogoHeight]{\IanLogoPath}\hspace{0.6em}%
  \fi
}

% Document label (appears next to small header logo)
\newcommand{\IanDocLabel}{}       % e.g. {BNR Refit – Phase 2}

% Large title-page logo (page 1 only)
\newcommand{\IanTitleLogoPath}{}  % e.g. {assets/brand/logo.png}
\newcommand{\IanTitleLogoWidth}{0.45\textwidth}

% ============================================================
% Title-page spacing controls (NO geometry changes)
% ============================================================

% Move the whole title block up (negative = higher)
% Tweak by eye: -1.0cm to -2.5cm are typical ranges.
\setlength{\droptitle}{-1.6cm}

% Space below the title logo
\newcommand{\IanTitleLogoVSpace}{0.8em}

% Space after the title
\newcommand{\IanPostTitleVSpace}{0.4em}

% Inject a large logo above the title (only if path provided)
\pretitle{%
  \begin{center}
  \ifx\IanTitleLogoPath\empty
  \else
    \includegraphics[width=\IanTitleLogoWidth]{\IanTitleLogoPath}\par\vspace{\IanTitleLogoVSpace}
  \fi
  \LARGE
}
\posttitle{\par\end{center}\vspace{\IanPostTitleVSpace}}

% ============================================================
% Header / Footer
% ============================================================

\pagestyle{fancy}
\fancyhf{} % clear everything

\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0pt}

% Left header: optional small logo + doc label
\lhead{\IanMaybeLogo\small\sffamily\IanDocLabel}

% Right header: current section title
\rhead{\small\sffamily\rightmark}

% Center footer: page X of Y
\cfoot{\small\sffamily Page \thepage\ of \pageref{LastPage}}

% Ensure rightmark updates cleanly with section titles
\renewcommand{\sectionmark}[1]{\markright{#1}}

% No running header/footer on title page (Pandoc uses 'plain' style there)
\fancypagestyle{plain}{
  \fancyhf{}
  \renewcommand{\headrulewidth}{0pt}
  \renewcommand{\footrulewidth}{0pt}
}

% End of tex/ian-house-style.tex
